def find_duplicates(lst):
    """
    Returns a list of all elements that appear more than once in `lst`,
    in the order they first appear.

    Args:
        lst: An iterable of hashable elements.

    Returns:
        A list of duplicate elements, preserving first-occurrence order.
    """
    seen = set()
    duplicates = []
    seen_duplicates = set()

    for item in lst:
        if item in seen:
            if item not in seen_duplicates:
                duplicates.append(item)
                seen_duplicates.add(item)
        else:
            seen.add(item)

    return duplicates

assert find_duplicates([1, 2, 3, 2, 1, 4]) == [2, 1]
assert find_duplicates(['a', 'b', 'a', 'c', 'b']) == ['a', 'b']
assert find_duplicates([1, 2, 3]) == []

def find_duplicates(lst):
    """
    Return a list of elements that appear more than once in lst,
    in the order of their first occurrence.

    Args:

        A list of duplicate elements, preserving first-occurrence order.
